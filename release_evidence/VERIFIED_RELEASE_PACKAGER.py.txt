from pathlib import Path

W=Path('/workspace/work');R=W/'BFT65B_SUBMISSION_CODE_VERIFIED_20261119';OLD=W/'BFT65B_SUBMISSION_CODE_RELEASE_20261118'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def write(p,x):p.write_text(x,encoding='utf-8')
records=[];relocations=[]
for p in sorted((R/'release_evidence').glob('C*/EXECUTION_RECORD.json')):
    v=read(p);records.append(v)
    for name,h in v['input_sha256_before'].items():
        snap=p.parent/'INPUT_SNAPSHOTS'/(h+'.txt');assert snap.is_file() and sha(snap)==h
    assert v['input_sha256_before']==v['input_sha256_after']
    for name,h in v['output_sha256'].items():
        if h is None:continue
        path=Path(name)
        if not path.is_absolute():assert sha(p.parent/name)==h;continue
        assert path.is_file() and sha(path)==h, name
        dest=R/'release_evidence/EXECUTED_OUTPUTS'/v['id']/path.name
        dest.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(path,dest)
        relocations.append([v['id'],name,dest.relative_to(R).as_posix(),h])
assert len(records)==19
counts={s:sum(v['status']==s for v in records) for s in ['PASS','BLOCKED','FAIL']}
